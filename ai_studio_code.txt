datasource db {
  provider = "postgresql"
  url      = env("DATABASE_URL")
}

generator client {
  provider = "prisma-client-js"
}

// ================= USER & RBAC =================

enum RoleType {
  SUPER_ADMIN
  ADMIN
  EDITOR
  PARTNER
  SUBSCRIBER
  REGISTERED
}

model User {
  id              String        @id @default(uuid())
  email           String        @unique
  passwordHash    String
  fullName        String?
  mobile          String?
  avatarUrl       String?
  role            RoleType      @default(REGISTERED)
  status          String        @default("ACTIVE") // ACTIVE, SUSPENDED, PENDING
  country         String?
  state           String?
  city            String?
  watchHours      Float         @default(0.0)
  totalSessions   Int           @default(0)
  lastLoginAt     DateTime?
  subscriptions   UserSubscription[]
  watchHistory    WatchHistory[]
  createdAt       DateTime      @default(now())
  updatedAt       DateTime      @updatedAt
}

// ================= VIDEO & MEDIA MANAGEMENT =================

enum VideoSourceType {
  FILE_UPLOAD
  M3U8_URL
  MP4_URL
  YOUTUBE_EMBED
}

model Video {
  id              String          @id @default(uuid())
  title           String
  slug            String          @unique
  shortDescription String?
  description     String?         @db.Text
  sourceType      VideoSourceType @default(FILE_UPLOAD)
  sourceUrl       String?         // Ingest file path or stream link
  hlsPlaybackUrl  String?         // Master .m3u8 CDN link
  thumbnailUrl    String?
  posterUrl       String?
  trailerUrl      String?
  durationSeconds Int             @default(0)
  
  // Categorization & Metadata
  categoryId      String?
  category        Category?       @relation(fields: [categoryId], references: [id])
  ageRating       String?         @default("13+")
  classification  String?         // U/A, PG, R
  languages       String[]        @default(["English"])
  subtitles       Json?           // Array of { lang: "en", url: "..." }
  
  // Playout & Scheduling
  isLive          Boolean         @default(false)
  isPPV           Boolean         @default(false)
  ppvPrice        Decimal?        @db.Decimal(10, 2)
  viewsCount      BigInt          @default(0)
  status          String          @default("READY") // PROCESSING, TRANSCODING, READY, FAILED
  
  seriesId        String?
  series          Series?         @relation(fields: [seriesId], references: [id])
  seasonNumber    Int?
  episodeNumber   Int?

  createdAt       DateTime        @default(now())
  updatedAt       DateTime        @updatedAt
}

model Category {
  id          String   @id @default(uuid())
  name        String   @unique
  slug        String   @unique
  description String?
  videos      Video[]
  createdAt   DateTime @default(now())
}

model Series {
  id          String   @id @default(uuid())
  title       String
  slug        String   @unique
  description String?  @db.Text
  genre       String?
  posterUrl   String?
  videos      Video[]  // Episodes
  createdAt   DateTime @default(now())
}

// ================= OTT APP BUILDS (JENKINS CI/CD) =================

model AppBuild {
  id          String   @id @default(uuid())
  platform    String   // ANDROID, ANDROID_TV, FIRE_TV, SAMSUNG_TV, LG_TV, ROKU, IOS, APPLE_TV
  buildFormat String   // APK, AAB, PKG
  appName     String
  packageName String
  versionCode Int      @default(1)
  versionName String   @default("1.0.0")
  buildStatus String   @default("QUEUED") // QUEUED, BUILDING, SUCCESS, FAILED
  downloadUrl String?
  jenkinsJobId String?
  createdAt   DateTime @default(now())
}

// ================= SUBSCRIPTIONS & MONETIZATION =================

model SubscriptionPlan {
  id              String   @id @default(uuid())
  name            String
  price           Decimal  @db.Decimal(10, 2)
  currency        String   @default("USD")
  interval        String   @default("MONTHLY") // MONTHLY, QUARTERLY, YEARLY
  maxDeviceLimit  Int      @default(2)
  allowedFeatures String[]
  createdAt       DateTime @default(now())
}

model UserSubscription {
  id        String           @id @default(uuid())
  userId    String
  user      User             @relation(fields: [userId], references: [id])
  planId    String
  status    String           @default("ACTIVE")
  startDate DateTime         @default(now())
  endDate   DateTime
}

model WatchHistory {
  id           String   @id @default(uuid())
  userId       String
  user         User     @relation(fields: [userId], references: [id])
  videoId      String
  watchedSecs  Int      @default(0)
  lastPosition Int      @default(0)
  updatedAt    DateTime @updatedAt
}
