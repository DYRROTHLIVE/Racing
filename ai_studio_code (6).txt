'use client';

import React, { useState, useEffect, useTransition } from 'react';
import { Search as SearchIcon, X, Clock, Film, Star } from 'lucide-react';
import Image from 'next/image';
import Link from 'next/link';
import { MediaItem } from '@/types/razy';

export const SearchExperience = () => {
  const [query, setQuery] = useState('');
  const [results, setResults] = useState<MediaItem[]>([]);
  const [history, setHistory] = useState<string[]>([]);
  const [isPending, startTransition] = useTransition();

  useEffect(() => {
    const saved = localStorage.getItem('razy_search_history');
    if (saved) setHistory(JSON.parse(saved));
  }, []);

  // Debounced API Search Query
  useEffect(() => {
    if (!query.trim()) {
      setResults([]);
      return;
    }

    const controller = new AbortController();
    const timeout = setTimeout(() => {
      startTransition(async () => {
        try {
          const res = await fetch(`/api/search?q=${encodeURIComponent(query)}`, {
            signal: controller.signal,
          });
          if (res.ok) {
            const data = await res.json();
            setResults(data.items || []);
          }
        } catch {
          // Request aborted or failed silently
        }
      });
    }, 280);

    return () => {
      clearTimeout(timeout);
      controller.abort();
    };
  }, [query]);

  const commitSearch = (term: string) => {
    setQuery(term);
    const updated = [term, ...history.filter((h) => h !== term)].slice(0, 8);
    setHistory(updated);
    localStorage.setItem('razy_search_history', JSON.stringify(updated));
  };

  const removeHistoryItem = (term: string, e: React.MouseEvent) => {
    e.stopPropagation();
    const updated = history.filter((h) => h !== term);
    setHistory(updated);
    localStorage.setItem('razy_search_history', JSON.stringify(updated));
  };

  return (
    <div className="max-w-5xl mx-auto px-4 py-8 pt-24 min-h-screen">
      {/* Search Input Bar */}
      <div className="relative flex items-center">
        <SearchIcon className="absolute left-4 w-5 h-5 text-[#A1A1B5]" />
        <input
          type="text"
          value={query}
          onChange={(e) => setQuery(e.target.value)}
          placeholder="Search movies, series, anime, actors..."
          className="w-full py-4 pl-12 pr-10 rounded-xl bg-[#171A26] border border-white/10 text-white placeholder-[#A1A1B5] focus:border-[#8B5CF6] focus:outline-none transition shadow-lg text-base"
        />
        {query && (
          <button
            onClick={() => setQuery('')}
            aria-label="Clear Search Input"
            className="absolute right-4 text-[#A1A1B5] hover:text-white"
          >
            <X className="w-5 h-5" />
          </button>
        )}
      </div>

      {/* Recent Searches */}
      {!query && history.length > 0 && (
        <div className="mt-8">
          <h3 className="text-sm font-semibold text-[#A1A1B5] uppercase tracking-wider mb-3">
            Recent Searches
          </h3>
          <div className="flex flex-wrap gap-2">
            {history.map((term) => (
              <span
                key={term}
                onClick={() => commitSearch(term)}
                className="flex items-center gap-2 px-3 py-1.5 rounded-full bg-[#10121B] border border-white/10 text-sm text-[#F8FAFC] cursor-pointer hover:border-purple-500 transition"
              >
                <Clock className="w-3.5 h-3.5 text-[#A1A1B5]" />
                {term}
                <X
                  onClick={(e) => removeHistoryItem(term, e)}
                  className="w-3.5 h-3.5 text-[#A1A1B5] hover:text-red-400"
                />
              </span>
            ))}
          </div>
        </div>
      )}

      {/* Results Grid / Empty State */}
      <div className="mt-8">
        {isPending ? (
          <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 gap-4">
            {[...Array(8)].map((_, i) => (
              <div key={i} className="aspect-[2/3] rounded-lg bg-white/5 animate-pulse" />
            ))}
          </div>
        ) : query && results.length === 0 ? (
          <div className="text-center py-20 text-[#A1A1B5]">
            <Film className="w-12 h-12 mx-auto mb-3 opacity-30" />
            <p className="text-lg font-semibold text-white">No titles found for &quot;{query}&quot;</p>
            <p className="text-sm mt-1">Check for spelling errors or try searching for another genre.</p>
          </div>
        ) : (
          <div className="grid grid-cols-2 sm:grid-cols-3 md:grid-cols-4 gap-4">
            {results.map((item) => (
              <Link
                key={item.id}
                href={`/title/${item.slug}`}
                onClick={() => commitSearch(query)}
                className="group relative rounded-lg overflow-hidden bg-[#10121B] border border-white/10 hover:border-purple-500 transition"
              >
                <div className="aspect-[2/3] relative">
                  <Image
                    src={item.posterUrl}
                    alt={item.title}
                    fill
                    className="object-cover group-hover:scale-105 transition duration-300"
                  />
                  {item.tier === 'vip' && (
                    <span className="absolute top-2 left-2 bg-cta-gradient text-[10px] font-bold px-1.5 py-0.5 rounded text-white shadow">
                      VIP
                    </span>
                  )}
                </div>
                <div className="p-3">
                  <h4 className="text-sm font-semibold text-white truncate">{item.title}</h4>
                  <div className="flex items-center justify-between text-xs text-[#A1A1B5] mt-1">
                    <span>{item.releaseYear}</span>
                    <span className="flex items-center gap-1 text-[#F59E0B]">
                      <Star className="w-3 h-3 fill-[#F59E0B]" />
                      {item.ratingScore}
                    </span>
                  </div>
                </div>
              </Link>
            ))}
          </div>
        )}
      </div>
    </div>
  );
};
