'use client';

import React, { useState } from 'react';
import Image from 'next/image';
import { Play, Plus, Share2, Star, Check } from 'lucide-react';
import { MediaItem, Episode } from '@/types/razy';

interface DetailsProps {
  item: MediaItem;
  onPlayEpisode: (ep: Episode) => void;
  onPlayMovie: (item: MediaItem) => void;
}

export const MediaDetailsView: React.FC<DetailsProps> = ({ item, onPlayEpisode, onPlayMovie }) => {
  const [selectedSeason, setSelectedSeason] = useState(1);
  const [copied, setCopied] = useState(false);

  const handleShare = async () => {
    if (navigator.share) {
      try {
        await navigator.share({
          title: item.title,
          text: item.synopsis,
          url: window.location.href,
        });
      } catch {
        // Share dismissed
      }
    } else {
      navigator.clipboard.writeText(window.location.href);
      setCopied(true);
      setTimeout(() => setCopied(false), 2000);
    }
  };

  const currentEpisodes = item.episodes?.filter((ep) => ep.seasonNumber === selectedSeason) || [];

  return (
    <div className="min-h-screen bg-[#08090F] pb-24">
      {/* Backdrop Header */}
      <div className="relative w-full h-[55vh] min-h-[400px]">
        <Image
          src={item.backdropUrl}
          alt={item.title}
          fill
          priority
          className="object-cover brightness-[0.6]"
        />
        <div className="absolute inset-0 bg-gradient-to-t from-[#08090F] via-[#08090F]/40 to-transparent" />
      </div>

      <div className="max-w-7xl mx-auto px-4 sm:px-6 lg:px-8 -mt-32 relative z-10">
        <div className="flex flex-col md:flex-row gap-8 items-start">
          {/* Poster Card */}
          <div className="w-44 sm:w-56 flex-shrink-0 rounded-xl overflow-hidden shadow-2xl border border-white/10 hidden md:block">
            <Image
              src={item.posterUrl}
              alt={item.title}
              width={224}
              height={336}
              className="w-full object-cover"
            />
          </div>

          {/* Details Content */}
          <div className="flex-1 space-y-4">
            <h1 className="text-3xl sm:text-4xl md:text-5xl font-black text-white">{item.title}</h1>

            <div className="flex flex-wrap items-center gap-3 text-sm text-[#A1A1B5]">
              <span className="flex items-center gap-1 text-[#F59E0B]">
                <Star className="w-4 h-4 fill-[#F59E0B]" />
                {item.ratingScore}
              </span>
              <span>{item.releaseYear}</span>
              <span className="border border-white/20 px-1.5 py-0.5 rounded text-xs">
                {item.contentRating}
              </span>
              <span>{item.genres.join(', ')}</span>
            </div>

            <p className="text-base text-[#F8FAFC]/80 max-w-3xl leading-relaxed">{item.synopsis}</p>

            <div className="flex items-center gap-4 pt-2">
              {item.kind === 'movie' ? (
                <button
                  onClick={() => onPlayMovie(item)}
                  className="flex items-center gap-2 px-6 py-3 rounded-lg font-bold bg-cta-gradient text-white shadow-ambient-purple hover:opacity-95 transition"
                >
                  <Play className="w-5 h-5 fill-white" />
                  <span>Play Movie</span>
                </button>
              ) : (
                <button
                  onClick={() => currentEpisodes[0] && onPlayEpisode(currentEpisodes[0])}
                  className="flex items-center gap-2 px-6 py-3 rounded-lg font-bold bg-cta-gradient text-white shadow-ambient-purple hover:opacity-95 transition"
                >
                  <Play className="w-5 h-5 fill-white" />
                  <span>Play S1 E1</span>
                </button>
              )}

              <button
                onClick={handleShare}
                aria-label="Share Title"
                className="p-3 rounded-lg razy-glass text-white hover:bg-white/10 transition"
              >
                {copied ? <Check className="w-5 h-5 text-green-400" /> : <Share2 className="w-5 h-5" />}
              </button>
            </div>
          </div>
        </div>

        {/* Series Episode Selector */}
        {item.kind === 'series' && item.episodes && (
          <div className="mt-16">
            <div className="flex items-center justify-between border-b border-white/10 pb-4 mb-6">
              <h2 className="text-xl font-bold text-white">Episodes</h2>
              {(item.seasonsCount || 1) > 1 && (
                <select
                  value={selectedSeason}
                  onChange={(e) => setSelectedSeason(Number(e.target.value))}
                  className="bg-[#171A26] border border-white/10 rounded-lg text-white px-3 py-1.5 text-sm"
                >
                  {[...Array(item.seasonsCount)].map((_, i) => (
                    <option key={i + 1} value={i + 1}>
                      Season {i + 1}
                    </option>
                  ))}
                </select>
              )}
            </div>

            <div className="grid grid-cols-1 md:grid-cols-2 gap-4">
              {currentEpisodes.map((ep) => (
                <div
                  key={ep.id}
                  onClick={() => onPlayEpisode(ep)}
                  className="group flex gap-4 p-3 rounded-xl bg-[#10121B] border border-white/5 hover:border-purple-500/50 cursor-pointer transition"
                >
                  <div className="relative w-36 aspect-video rounded-lg overflow-hidden flex-shrink-0 bg-neutral-800">
                    <Image
                      src={ep.thumbnailUrl}
                      alt={ep.title}
                      fill
                      className="object-cover group-hover:scale-105 transition"
                    />
                    <div className="absolute inset-0 bg-black/40 flex items-center justify-center opacity-0 group-hover:opacity-100 transition">
                      <Play className="w-6 h-6 fill-white text-white" />
                    </div>
                  </div>
                  <div className="flex-1 overflow-hidden">
                    <div className="flex items-center justify-between text-xs text-[#A1A1B5]">
                      <span>Episode {ep.episodeNumber}</span>
                      <span>{Math.floor(ep.duration / 60)}m</span>
                    </div>
                    <h4 className="text-sm font-semibold text-white mt-1 truncate">{ep.title}</h4>
                    <p className="text-xs text-[#A1A1B5] line-clamp-2 mt-1">{ep.synopsis}</p>
                  </div>
                </div>
              ))}
            </div>
          </div>
        )}

        {/* Cast & Crew Carousel */}
        <div className="mt-16">
          <h2 className="text-xl font-bold text-white mb-4">Cast &amp; Crew</h2>
          <div className="flex gap-4 overflow-x-auto no-scrollbar pb-4">
            {item.cast.map((person) => (
              <div key={person.id} className="flex flex-col items-center flex-shrink-0 w-24">
                <div className="w-20 h-20 rounded-full overflow-hidden border border-white/10 relative">
                  <Image src={person.avatarUrl} alt={person.name} fill className="object-cover" />
                </div>
                <span className="text-xs font-semibold text-white mt-2 text-center truncate w-full">
                  {person.name}
                </span>
                <span className="text-[10px] text-[#A1A1B5] text-center truncate w-full">
                  {person.role}
                </span>
              </div>
            ))}
          </div>
        </div>
      </div>
    </div>
  );
};
