'use client';

import React, { useEffect, useState } from 'react';
import { motion, AnimatePresence } from 'framer-motion';

export const SplashScreen: React.FC<{ onComplete: () => void }> = ({ onComplete }) => {
  const [stage, setStage] = useState<'animating' | 'done'>('animating');

  const playCinematicBoom = () => {
    try {
      const AudioCtx = window.AudioContext || (window as unknown as { webkitAudioContext: typeof AudioContext }).webkitAudioContext;
      if (!AudioCtx) return;
      const ctx = new AudioCtx();
      
      const osc = ctx.createOscillator();
      const gain = ctx.createGain();
      
      osc.type = 'sawtooth';
      osc.frequency.setValueAtTime(55, ctx.currentTime);
      osc.frequency.exponentialRampToValueAtTime(28, ctx.currentTime + 1.6);

      gain.gain.setValueAtTime(0.35, ctx.currentTime);
      gain.gain.exponentialRampToValueAtTime(0.001, ctx.currentTime + 1.8);

      osc.connect(gain);
      gain.connect(ctx.destination);
      osc.start();
      osc.stop(ctx.currentTime + 1.8);
    } catch {
      // Audio autoplay policy fallback
    }
  };

  useEffect(() => {
    playCinematicBoom();
    const timer = setTimeout(() => {
      setStage('done');
      onComplete();
    }, 1800);
    return () => clearTimeout(timer);
  }, [onComplete]);

  return (
    <AnimatePresence>
      {stage === 'animating' && (
        <motion.div
          className="fixed inset-0 z-50 flex flex-col items-center justify-center bg-[#08090F]"
          initial={{ opacity: 1 }}
          exit={{ opacity: 0, scale: 1.05 }}
          transition={{ duration: 0.45, ease: 'easeInOut' }}
        >
          {/* Subtle Ambient Glow */}
          <div className="absolute inset-0 bg-ambient-glow pointer-events-none" />

          {/* Logo Reveal */}
          <motion.div
            initial={{ scale: 0.85, opacity: 0 }}
            animate={{ scale: 1, opacity: 1 }}
            transition={{ duration: 0.8, ease: [0.16, 1, 0.3, 1] }}
            className="relative flex items-center gap-1 text-5xl md:text-6xl font-black tracking-tighter"
          >
            <span className="text-white">R</span>
            <span className="bg-gradient-to-r from-[#8B5CF6] to-[#EC4899] bg-clip-text text-transparent">
              Azy
            </span>
            <div className="absolute -inset-4 bg-purple-500/10 blur-2xl rounded-full" />
          </motion.div>

          <motion.div
            initial={{ width: 0 }}
            animate={{ width: 140 }}
            transition={{ delay: 0.3, duration: 1.2, ease: 'easeInOut' }}
            className="h-1 mt-6 rounded-full bg-gradient-to-r from-[#7C3AED] to-[#DB2777]"
          />
        </motion.div>
      )}
    </AnimatePresence>
  );
};
